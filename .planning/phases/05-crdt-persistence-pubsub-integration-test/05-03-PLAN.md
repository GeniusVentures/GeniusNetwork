        // --- Sub-case A: Tampered child signature ---

            SGTransaction::RegistrationMetadata metadata;

            auto tx = RegistrationTransaction::New(main_address, 3, metadata, FillDAGStructForTest());
            // Build valid tx first, then tamper a byte in the signature
            tx->MakeSignature(*child_account);  // Assume we have child_account access
            auto serialized = tx->SerializeByteVector();
            // Tamper: flip a byte in the signature portion (last 64 bytes of serialized data)
            if (serialized.size() > 64)
            {
                serialized[serialized.size() - 10] ^= 0xFF;
            }

            crdt::pb::Element element;
            element.set_key(reg_key);
            element.set_value(std::string(serialized.begin(), serialized.end()));

    The sub-case C (non-monotonic sequence) requires injecting a valid CRDT element first to establish sequence=5 in the reg/ CRDT, then injecting sequence=3 to trigger the rejection. To persist the base element, use the TransactionManager's CRDT transaction path or directly write via `globaldb_m->Put(reg_key, value)` followed by `Commit()`.     The executor should choose the simplest approach that correctly exercises gate (d) rejection — either the direct-CRDT-write approach or the double-RegisterChild-then-inject approach.
