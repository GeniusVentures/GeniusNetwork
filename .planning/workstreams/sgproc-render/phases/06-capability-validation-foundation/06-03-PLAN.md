**B) ProcessingManager.cpp changes:**

1. In `Init()`, after all `RegisterProcessorFactory` and `RegisterPassProcessorFactory` calls (line ~401 in current code — after the `RegisterPassProcessorFactory(PassType::RENDER, ...)` call), add:

```cpp
// Build capability snapshot after all executors are registered (D-01, D-09)
m_capabilityValidator = std::make_unique<CapabilityValidator>();
m_capabilityValidator->BuildSnapshot(m_passFactories,
    m_processorFactories.size());
```

This runs BEFORE `CheckProcessValidity()` because `BuildSnapshot` doesn't depend on the parsed JSON — it only needs the registered factory maps.

2. Implement `ProcessingManager::CanExecute()`:
```cpp
void ProcessingManager::CanExecute(const sgns::SgnsPass& pass,
                                    sgns::sgprocessing::CanExecuteCallback callback)
{
    if (!m_capabilityValidator)
    {
        // Defensive: if BuildSnapshot wasn't called (shouldn't happen after Init),
        // report as not executable with a RESOURCE unmet reason.
        CanExecuteResult result;
        result.executable = false;
        result.unmet.push_back({UnmetRequirementCategory::RESOURCE,
            "CapabilityValidator not initialized"});
        callback(result);
        return;
    }
    m_capabilityValidator->CanExecute(pass, std::move(callback));
}
```

```cpp
class CapabilityValidatorTest : public ::testing::Test {
protected:
    void SetUp() override {
        validator = std::make_unique<CapabilityValidator>();
    }
    
    // Helper: build a mock snapshot with custom Vulkan limits
    CapabilitySnapshot MakeMockSnapshot(
        VkPhysicalDeviceLimits limits = {},
        VkPhysicalDeviceType deviceType = VK_PHYSICAL_DEVICE_TYPE_DISCRETE_GPU,
        VkDeviceSize largestHeap = 1024ULL * 1024 * 1024, // 1GB
        std::vector<ExecutorCapability> caps = {},
        uint64_t diskBytes = 10ULL * 1024 * 1024 * 1024) // 10GB
    {
        CapabilitySnapshot snap;
        snap.vulkanProps.limits = limits;
        snap.vulkanProps.deviceType = deviceType;
        // Set one device-local heap
        snap.memProps.memoryHeapCount = 2;
        snap.memProps.memoryHeaps[0] = {largestHeap, VK_MEMORY_HEAP_DEVICE_LOCAL_BIT};
        snap.memProps.memoryHeaps[1] = {256*1024*1024, 0}; // non-device-local
        snap.executorCaps = std::move(caps);
        snap.availableDiskBytes = diskBytes;
        // Compute a fake identityHash (sha256 of deviceID+vendorID)
        uint32_t deviceID = 0x1234;
        snap.identityHash = sgns::sgprocmanagersha::sha256(&deviceID, sizeof(deviceID));
        return snap;
    }

    std::unique_ptr<CapabilityValidator> validator;
};
```
